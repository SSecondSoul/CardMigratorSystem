<script>
function __sanComputed_getX_0(index, dataPoints, width) {
    if (dataPoints.length <= 1) return width / 2;
    return (index / (dataPoints.length - 1)) * width;
  }
function __sanComputed_getY_1(value, maxValue, minValue, height) {
    const range = maxValue - minValue;
    if (range === 0) return height / 2;
    return height - ((value - minValue) / range) * height;
  }

const RealTimeChart = san.defineComponent({
  dataTypes: {"data": san.DataTypes.any, "height": san.DataTypes.any, "color": san.DataTypes.any, "loading": san.DataTypes.any},

  name: 'RealTimeChart',
  template: `<div class="chart-wrapper {{ loading ? 'loading' : '' }}">
    <svg width="100%" height="{{ height }}" viewBox="0 0 {{ width }} {{ height }}">
      <defs>
        <linearGradient id="{{ gradientId }}" x1="0%" y1="0%" x2="0%" y2="100%">
          <stop offset="0%" stop-color="{{ color }}" stop-opacity="0.3"/>
          <stop offset="100%" stop-color="{{ color }}" stop-opacity="0.0"/>
        </linearGradient>
      </defs>
      <g s-for="i in 4 trackby i">
        <line
          x1="0" y1="{{ i * (height / 4) }}"
          x2="{{ width }}" y2="{{ i * (height / 4) }}"
          stroke="#e2e8f0" stroke-width="0.5" stroke-dasharray="3,3"
        />
      </g>
      <polygon points="{{ areaPoints }}" fill="{{ 'url(#' + gradientId + ')' }}" />
      <polyline points="{{ linePoints }}" fill="none" stroke="{{ color }}" stroke-width="2" stroke-linecap="round" />
      <circle
        s-for="point, idx in dataPoints trackby idx"
        cx="{{ getX(idx, dataPoints, width) }}" cy="{{ getY(point, maxValue, minValue, height) }}" r="3"
        fill="{{ color }}" stroke="white" stroke-width="2"
      />
    </svg>
  </div>`,
  initData() {
    return {
      width: 600,
      data: [],
      height: 250,
      color: '#3b82f6',
      loading: false
    };
  },
  computed: {
    gradientId() {
      const color = this.data.get('color');
      return `gradient-${color.replace('#', '')}`;
    },
    dataPoints() {
      const data = this.data.get('data');
      return data.map(d => d.value);
    },
    maxValue() {
      const points = this.data.get('dataPoints');
      return Math.max(...points, 100);
    },
    minValue() {
      const points = this.data.get('dataPoints');
      return Math.min(...points, 0);
    },
    linePoints() {
      const points = this.data.get('dataPoints');
      const height = this.data.get('height');
      const width = this.data.get('width');
      return points.map((val, idx) => `${__sanComputed_getX_0(idx, points, width)},${__sanComputed_getY_1(val, this.data.get('maxValue'), this.data.get('minValue'), height)}`).join(' ');
    },
    areaPoints() {
      const points = this.data.get('dataPoints');
      const height = this.data.get('height');
      const width = this.data.get('width');
      const areaPoints = points.map((val, idx) => `${__sanComputed_getX_0(idx, points, width)},${__sanComputed_getY_1(val, this.data.get('maxValue'), this.data.get('minValue'), height)}`);
      return `0,${height} ${areaPoints.join(' ')} ${width},${height}`;
    }
  },
  attached() {
    this.updateWidth();
    window.addEventListener('resize', () => this.updateWidth());
  },
  detached() {
    window.removeEventListener('resize', () => this.updateWidth());
  },
  updateWidth() {
    if (this.el) {
      this.data.set('width', this.el.clientWidth - 20);
    }
  },
  getX(index, dataPoints, width) {
    if (dataPoints.length <= 1) return width / 2;
    return (index / (dataPoints.length - 1)) * width;
  },
  getY(value, maxValue, minValue, height) {
    const range = maxValue - minValue;
    if (range === 0) return height / 2;
    return height - ((value - minValue) / range) * height;
  }
});

    return (count / maxCount) * 280;

      }
    ];
    this.data.set('activeAlerts', activeAlerts);
  },
  initLogs() {
    const levels = ['info', 'warning', 'error'];
    const messages = [
      '用户登录成功', '数据库查询慢', 'API请求失败', '缓存命中',
      '服务重启', '配置更新', '磁盘空间不足', '网络延迟增加'
    ];
    const sources = ['API Gateway', 'Auth Service', 'Database', 'Cache', 'Worker'];

    const systemLogs = [];
    for (let i = 0; i < 50; i++) {
      const level = levels[Math.floor(Math.random() * levels.length)];
      systemLogs.push({
        id: i,
        level,
        message: messages[Math.floor(Math.random() * messages.length)],
        source: sources[Math.floor(Math.random() * sources.length)],
        timestamp: Date.now() - Math.random() * 86400000
      });
    }
    systemLogs.sort((a, b) => b.timestamp - a.timestamp);
    this.data.set('systemLogs', systemLogs);
  },
  initServices() {
    const services = [
      { id: 1, name: 'API Gateway', status: 'healthy', responseTime: 45, availability: 99.9 },
      { id: 2, name: 'Auth Service', status: 'healthy', responseTime: 67, availability: 99.8 },
      { id: 3, name: 'User Service', status: 'warning', responseTime: 234, availability: 95.2 },
      { id: 4, name: 'Payment Service', status: 'healthy', responseTime: 89, availability: 99.5 },
      { id: 5, name: 'Database', status: 'healthy', responseTime: 12, availability: 99.99 },
      { id: 6, name: 'Cache Service', status: 'warning', responseTime: 156, availability: 97.3 }
    ];
    this.data.set('services', services);
  },
  startDataSimulation() {
    const dataGenerator = setInterval(() => {
      this.simulateDataUpdate();
    }, 2000);
    this.data.set('dataGenerator', dataGenerator);
  },
  stopDataSimulation() {
    const dataGenerator = this.data.get('dataGenerator');
    if (dataGenerator) {
      clearInterval(dataGenerator);
      this.data.set('dataGenerator', null);
    }
  },
  simulateDataUpdate() {
    const kpiData = this.data.get('kpiData');
    const cpuHistory = this.data.get('cpuHistory');
    const memoryHistory = this.data.get('memoryHistory');

    // Update KPI data
    const newKpiData = [...kpiData];
    newKpiData[0].value = Math.max(5, Math.min(95, kpiData[0].value + (Math.random() - 0.5) * 3));
    newKpiData[1].value = Math.max(30, Math.min(90, kpiData[1].value + (Math.random() - 0.5) * 2));
    newKpiData[2].value = Math.max(95, Math.min(100, kpiData[2].value + (Math.random() - 0.5) * 0.5));
    newKpiData[3].value = Math.max(50, Math.min(500, kpiData[3].value + (Math.random() - 0.5) * 10));

    // Update chart data
    const newCpuValue = 30 + Math.random() * 50;
    const newMemoryValue = 40 + Math.random() * 40;

    const newCpuHistory = [...cpuHistory, { time: Date.now(), value: newCpuValue }];
    const newMemoryHistory = [...memoryHistory, { time: Date.now(), value: newMemoryValue }];

    if (newCpuHistory.length > 60) newCpuHistory.shift();
    if (newMemoryHistory.length > 60) newMemoryHistory.shift();

    // Check and generate alerts
    if (newCpuValue > 85 && Math.random() > 0.7) {
      this.addAlert('CPU使用率异常', `CPU使用率达到 ${newCpuValue.toFixed(1)}%`, 'critical');
    }
    if (newMemoryValue > 85 && Math.random() > 0.8) {
      this.addAlert('内存使用率过高', `内存使用率达到 ${newMemoryValue.toFixed(1)}%`, 'warning');
    }

    // Add random log
    if (Math.random() > 0.7) {
      const levels = ['info', 'warning', 'error'];
      const level = levels[Math.floor(Math.random() * levels.length)];
      this.addLog(level, `系统运行状态更新`, 'Monitor');
    }

    this.data.set('kpiData', newKpiData);
    this.data.set('cpuHistory', newCpuHistory);
    this.data.set('memoryHistory', newMemoryHistory);
  },
  refreshAllData() {
    return new Promise((resolve) => {
      this.data.set('isChartLoading', true);
      this.data.set('isRefreshingServices', true);

      this.showNotification('正在刷新数据...', 'info');

      setTimeout(() => {
        this.initRequestStats();
        this.initServices();

        this.data.set('isChartLoading', false);
        this.data.set('isRefreshingServices', false);
        this.showNotification('数据刷新完成', 'success');
        resolve();
      }, 1000);
    });
  },
  refreshServices() {
    this.refreshAllData();
  },
  startAutoRefresh() {
    this.stopAutoRefresh();
    const interval = this.data.get('settings').refreshInterval * 1000;
    const refreshTimer = setInterval(() => {
      this.refreshAllData();
    }, interval);
    this.data.set('refreshTimer', refreshTimer);
  },
  stopAutoRefresh() {
    const refreshTimer = this.data.get('refreshTimer');
    if (refreshTimer) {
      clearInterval(refreshTimer);
      this.data.set('refreshTimer', null);
    }
  },
  startTimeUpdate() {
    const timeTimer = setInterval(() => {
      this.updateDateTime();
    }, 1000);
    this.data.set('timeTimer', timeTimer);
  },
  stopTimeUpdate() {
    const timeTimer = this.data.get('timeTimer');
    if (timeTimer) {
      clearInterval(timeTimer);
      this.data.set('timeTimer', null);
    }
  },
  updateDateTime() {
    const now = new Date();
    this.data.set('currentDate', now.toLocaleDateString('zh-CN'));
    this.data.set('currentTime', now.toLocaleTimeString('zh-CN'));
  },
  changePeriod(period) {
    this.data.set('selectedPeriod', period);
    this.showNotification(`切换到 ${period} 视图`, 'info');
  },
  addAlert(title, description, severity) {
    const alertIdCounter = this.data.get('alertIdCounter');
    const newAlert = {
      id: ++alertIdCounter,
      title,
      description,
      severity,
      timestamp: Date.now()
    };
    const activeAlerts = this.data.get('activeAlerts');
    const newActiveAlerts = [newAlert, ...activeAlerts];
    this.data.set('alertIdCounter', alertIdCounter);
    this.data.set('activeAlerts', newActiveAlerts);
    this.showNotification(`新告警: ${title}`, 'warning');

    const settings = this.data.get('settings');
    if (settings.desktopNotifications && Notification.permission === 'granted') {
      new Notification(title, { body: description });
    }
  },
  acknowledgeAlert(alertId) {
    const activeAlerts = this.data.get('activeAlerts');
    const alertIndex = activeAlerts.findIndex(a => a.id === alertId);
    if (alertIndex !== -1) {
      const newActiveAlerts = [...activeAlerts];
      newActiveAlerts[alertIndex] = { ...newActiveAlerts[alertIndex], acknowledged: true };
      this.data.set('activeAlerts', newActiveAlerts);
      this.showNotification(`已确认告警: ${newActiveAlerts[alertIndex].title}`, 'info');
    }
  },
  resolveAlert(alertId) {
    const activeAlerts = this.data.get('activeAlerts');
    const alertIndex = activeAlerts.findIndex(a => a.id === alertId);
    if (alertIndex !== -1) {
      const resolved = activeAlerts[alertIndex];
      const newActiveAlerts = [...activeAlerts.slice(0, alertIndex), ...activeAlerts.slice(alertIndex + 1)];
      const alertHistory = this.data.get('alertHistory');
      const newAlertHistory = [...alertHistory, { ...resolved, resolvedAt: Date.now() }];
      this.data.set('activeAlerts', newActiveAlerts);
      this.data.set('alertHistory', newAlertHistory);
      this.showNotification(`已解决告警: ${resolved.title}`, 'success');
    }
  },
  clearAlerts() {
    this.data.set('activeAlerts', []);
    this.showNotification('已清空所有告警', 'info');
  },
  addLog(level, message, source) {
    const newLog = {
      id: Date.now(),
      level,
      message,
      source,
      timestamp: Date.now()
    };
    const systemLogs = this.data.get('systemLogs');
    const newSystemLogs = [newLog, ...systemLogs];
    if (newSystemLogs.length > 500) {
      this.data.set('systemLogs', newSystemLogs.slice(0, 500));
    } else {
      this.data.set('systemLogs', newSystemLogs);
    }
  },
  filterLogs() {
    this.showNotification(`筛选日志: ${this.data.get('logFilter')}`, 'info');
  },
  clearLogs() {
    this.data.set('systemLogs', []);
    this.showNotification('日志已清空', 'info');
  },
  onKPIClick(kpi) {
    this.showNotification(`查看详情: ${kpi.title}`, 'info');
    this.addLog('info', `查看KPI指标: ${kpi.title}`, 'Dashboard');
  },
  onBarClick(data) {
    this.showNotification(`查看时段: ${data.hour} 请求量 ${data.count}`, 'info');
  },
  onLogClick() {
    const logItem = this.event.target.closest('.log-item');
    if (logItem) {
      const id = parseInt(logItem.getAttribute('data-id'));
      if (!isNaN(id)) {
        const logs = this.data.get('filteredLogs');
        const log = logs.find(l => l.id === id);
        if (log) {
          this.showNotification(`日志详情: ${log.message}`, 'info');
        }
      }
    }
  },
  onServiceClick() {
    const serviceCard = this.event.target.closest('.service-card');
    if (serviceCard) {
      const id = parseInt(serviceCard.getAttribute('data-id'));
      if (!isNaN(id)) {
        const services = this.data.get('services');
        const service = services.find(s => s.id === id);
        if (service) {
          this.showNotification(`服务详情: ${service.name} (${service.status})`, 'info');
        }
      }
    }
  },
  exportReport() {
    const report = {
      exportTime: new Date().toISOString(),
      kpi: this.data.get('kpiData'),
      alerts: this.data.get('activeAlerts'),
      services: this.data.get('services'),
      summary: {
        totalAlerts: this.data.get('activeAlerts').length,
        healthyServices: this.data.get('services').filter(s => s.status === 'healthy').length,
        avgCpu: this.data.get('kpiData')[0].value,
        avgMemory: this.data.get('kpiData')[1].value
      }
    };

    const blob = new Blob([JSON.stringify(report, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `monitoring_report_${Date.now()}.json`;
    a.click();
    URL.revokeObjectURL(url);
    this.showNotification('报告导出成功', 'success');
  },
  saveSettings(settings) {
    this.stopAutoRefresh();
    this.data.set('settings', { ...settings });
    if (settings.autoRefresh) {
      this.startAutoRefresh();
    }
    this.data.set('showSettings', false);
    this.showNotification('设置已保存', 'success');
  },
  toggleTheme() {
    const isDarkMode = !this.data.get('isDarkMode');
    this.data.set('isDarkMode', isDarkMode);
    this.showNotification(`已切换到${isDarkMode ? '暗色' : '亮色'}模式`, 'info');
  },
  showNotification(message, type) {
    const timer = this.data.get('notificationTimer');
    if (timer) {
      clearTimeout(timer);
    }
    this.data.set('notification', {
      show: true,
      message,
      type
    });
    const notificationTimer = setTimeout(() => {
      this.hideNotification();
      this.data.set('notificationTimer', null);
    }, 3000);
    this.data.set('notificationTimer', notificationTimer);
  },
  hideNotification() {
    const notification = this.data.get('notification');
    this.data.set('notification', { ...notification, show: false });
  },
  handleModalClick(e) {
    if (e.target === e.currentTarget) {
      this.data.set('showSettings', false);
    }
  },
  showSettingsHandler() {
    this.data.set('showSettings', true);
  },
  hideSettingsHandler() {
    this.data.set('showSettings', false);
  }
});
</script>
